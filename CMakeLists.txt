cmake_minimum_required(VERSION 3.16)
project(cross_mcu_sensor_framework LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sensor_framework src/filters.cpp src/temperature_sensor.cpp src/application_adapters.cpp)
target_include_directories(sensor_framework PUBLIC include)
add_executable(sensor_demo examples/host_demo.cpp)
target_link_libraries(sensor_demo PRIVATE sensor_framework)
enable_testing()
add_executable(sensor_tests tests/sensor_tests.cpp)
target_link_libraries(sensor_tests PRIVATE sensor_framework)
add_test(NAME sensor_tests COMMAND sensor_tests)
